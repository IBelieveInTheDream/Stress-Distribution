Choice = input('Enter C for Compound or T for Thick cylinder: ','s');

if strcmpi(Choice,'T')

    a=input('Inner radius: ');
    b=input('Outer radius: ');
    Pi=input('Internal pressure: ');
    Po=input('Outer pressure: ');

    A=(Pi*a^2-Po*b^2)/(b^2-a^2);
    B=a^2*b^2*(Pi-Po)/(b^2-a^2);

    r=linspace(a,b,200);
    Sr = A-B./r.^2;
    Sh = A+B./r.^2;
    Sr1=A-B/a^2;
    Sh1=A+B/a^2;

    fprintf('Radial stress at r=a:\n'); disp(Sr1);
    fprintf('Hoop stress at r=a:\n'); disp(Sh1);

    Sr2=A-B/b^2;
    Sh2=A+B/b^2;

    fprintf('Radial stress at r=b:\n'); disp(Sr2);
    fprintf('Hoop stress at r=b:\n'); disp(Sh2);

    plot(r,Sr,'b',r,Sh,'r','LineWidth',2);
    xlabel('Radius (mm)'); ylabel('Stress (MPa)');
    legend('Radial','Hoop'); grid on;
    title('Thick Cylinder');


elseif strcmpi(Choice,'C')

    a=input('Inner radius: ');
    b=input('Interface radius: ');
    c=input('Outer radius: ');
    p=input('Shrinkage pressure: ');
    q=input('Internal pressure: ');

    r=[linspace(a,b,250),linspace(b,c,250)];
    D1=b^2-a^2;
    D2=c^2-b^2;

    % Shrinkage pressure constants
    A1=-p*b^2/D1; B1=-p*a^2*b^2/D1;
    A2=p*b^2/D2;  B2=p*b^2*c^2/D2;

    % Internal pressure interface pressure
    Pint=a^2*q*D2/(b^2*(c^2-a^2));

    A3=(q*a^2-Pint*b^2)/D1;
    B3=a^2*b^2*(q-Pint)/D1;
    A4=Pint*b^2/D2;
    B4=Pint*b^2*c^2/D2;

    % Stress calculation in both cylinders
    idx1=r<=b;
    idx2=r>b;

    Sr_s=zeros(size(r)); Sh_s=Sr_s;
    Sr_i=Sr_s; Sh_i=Sr_s;

    Sr_s(idx1)=A1-B1./r(idx1).^2;
    Sh_s(idx1)=A1+B1./r(idx1).^2;
    Sr_s(idx2)=A2-B2./r(idx2).^2;
    Sh_s(idx2)=A2+B2./r(idx2).^2;

    Sr_i(idx1)=A3-B3./r(idx1).^2;
    Sh_i(idx1)=A3+B3./r(idx1).^2;
    Sr_i(idx2)=A4-B4./r(idx2).^2;
    Sh_i(idx2)=A4+B4./r(idx2).^2;

    % Total stresses
    Sr_t=Sr_s+Sr_i;
    Sh_t=Sh_s+Sh_i;

    % Surface values: a, b(inner), b(outer), c
    R=[a b b c];

    As=[A1 A1 A2 A2];
    Bs=[B1 B1 B2 B2];
    Ai=[A3 A3 A4 A4];
    Bi=[B3 B3 B4 B4];

    Sr_surf_s=As-Bs./R.^2;
    Sh_surf_s=As+Bs./R.^2;

    Sr_surf_i=Ai-Bi./R.^2;
    Sh_surf_i=Ai+Bi./R.^2;

    Sr_surf_t=Sr_surf_s+Sr_surf_i;
    Sh_surf_t=Sh_surf_s+Sh_surf_i;

    % Display all results
    Names={'Inner surface (a)       ','Inner cylinder outer (b)', ...
           'Outer cylinder inner (b)','Outer surface (c)        '};

    fprintf('\nInterface pressure = %.2f MPa\n',Pint);

    Results={Sr_surf_s,Sh_surf_s;
             Sr_surf_i,Sh_surf_i;
             Sr_surf_t,Sh_surf_t};

    Labels={'SHRINKAGE PRESSURE','INTERNAL PRESSURE','TOTAL STRESSES'};

    for j=1:3
        fprintf('\n--- %s ---\n',Labels{j});
        fprintf('Surface\t\t\tRadial\t\tHoop\n');
        for k=1:4
            fprintf('%s\t%.2f\t\t%.2f\n', ...
                Names{k},Results{j,1}(k),Results{j,2}(k));
        end
    end

    % Plot
    plot(r,Sr_s,'b','LineWidth',1.5);
    hold on;
    plot(r,Sh_s,'r','LineWidth',1.5);
    plot(r,Sr_i,'g','LineWidth',1.5);
    plot(r,Sh_i,'y','LineWidth',1.5);
    plot(r,Sr_t,'k','LineWidth',2.5);
    plot(r,Sh_t,'m','LineWidth',2.5);

    xlabel('Radius (mm)');
    ylabel('Stress (MPa)');
    title('Stress Distribution in Compound Cylinder');
    legend('Radial shrinkage','Hoop shrinkage', ...
           'Radial internal','Hoop internal', ...
           'Total radial','Total hoop');
    grid on;
    hold off;

else
    disp('Invalid Input');
end
